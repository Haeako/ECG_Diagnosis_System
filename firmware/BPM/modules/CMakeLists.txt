cmake_minimum_required(VERSION 3.16)
set(ECG_BPM_ALGORITHM "fast_qrs" CACHE STRING "One selected standalone BPM QRS module")
set(ECG_QRS_KERNELS "${CMAKE_CURRENT_LIST_DIR}/../../esp32/main/components/qrs_detector")
set(ECG_QRS_NAMES pan elgendi fast_qrs panpp wim_nguyen wim_elgendi wim_pantompkins
    wim_hamilton wim_arteaga_falconi wim_shaik wim_zhai wim_xu)
if(NOT ECG_BPM_ALGORITHM IN_LIST ECG_QRS_NAMES)
    message(FATAL_ERROR "Unknown ECG_BPM_ALGORITHM: ${ECG_BPM_ALGORITHM}")
endif()

function(ecg_qrs_describe name)
    if(name STREQUAL "pan")
        set(base pan_pyecg)
        set(state pan_pyecg_workspace_t)
        set(detect pan_pyecg_detect_360)
    elseif(name STREQUAL "elgendi")
        set(base elgendi_pyecg)
        set(state elgendi_pyecg_workspace_t)
        set(detect elgendi_pyecg_detect_360)
    elseif(name STREQUAL "fast_qrs")
        set(base fast_qrs_port)
        set(state fast_qrs_workspace_t)
        set(detect fast_qrs_detect_360)
    elseif(name STREQUAL "panpp")
        set(base pan_tompkins_plus_plus_port)
        set(state panpp_workspace_t)
        set(detect panpp_detect_360)
    elseif(name STREQUAL "wim_pantompkins")
        set(base ${name})
        set(state wim_pt_workspace_t)
        set(detect wim_pt_detect_360)
    elseif(name STREQUAL "wim_arteaga_falconi")
        set(base ${name})
        set(state wim_af_workspace_t)
        set(detect wim_af_detect_360)
    else()
        set(base ${name})
        set(state ${name}_workspace_t)
        set(detect ${name}_detect_360)
    endif()
    if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/${base}.c")
        set(source "${CMAKE_CURRENT_LIST_DIR}/${base}.c")
    else()
        set(source "${ECG_QRS_KERNELS}/${base}.c")
    endif()
    set(QRS_SOURCE ${source} PARENT_SCOPE)
    set(QRS_HEADER ${base}.h PARENT_SCOPE)
    set(QRS_STATE ${state} PARENT_SCOPE)
    set(QRS_DETECT ${detect} PARENT_SCOPE)
endfunction()

if(ESP_PLATFORM)
    ecg_qrs_describe(${ECG_BPM_ALGORITHM})
    idf_component_register(SRCS qrs_module.c "${QRS_SOURCE}"
        INCLUDE_DIRS "." "${ECG_QRS_KERNELS}")
    target_compile_features(${COMPONENT_LIB} PRIVATE c_std_11)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE
        ECG_QRS_HEADER="${QRS_HEADER}" ECG_QRS_STATE=${QRS_STATE} ECG_QRS_DETECT=${QRS_DETECT})
else()
    project(ecg_bpm_qrs_modules C)
    foreach(name IN LISTS ECG_QRS_NAMES)
        option(ECG_BPM_BUILD_${name} "Build ${name} module" ON)
        if(ECG_BPM_BUILD_${name})
            ecg_qrs_describe(${name})
            add_library(ecg_qrs_${name} STATIC qrs_module.c "${QRS_SOURCE}")
            target_compile_features(ecg_qrs_${name} PRIVATE c_std_11)
            target_include_directories(ecg_qrs_${name} PUBLIC "." PRIVATE "${ECG_QRS_KERNELS}")
            target_compile_definitions(ecg_qrs_${name} PRIVATE
                ECG_QRS_HEADER="${QRS_HEADER}" ECG_QRS_STATE=${QRS_STATE} ECG_QRS_DETECT=${QRS_DETECT})
            target_link_libraries(ecg_qrs_${name} PUBLIC m)
        endif()
    endforeach()
    if(TARGET ecg_qrs_${ECG_BPM_ALGORITHM})
        add_library(ecg_bpm_selected ALIAS ecg_qrs_${ECG_BPM_ALGORITHM})
    endif()
endif()
