cmake_minimum_required(VERSION 3.16)
set(ECG_BUILD_PROFILE af-extra-trees CACHE STRING "Selected AFib benchmark")

if(NOT ECG_BUILD_PROFILE IN_LIST _allowed_profiles)
    message(FATAL_ERROR "Unsupported AFib profile: ${ECG_BUILD_PROFILE}")
endif()
set(ENV{ECG_BUILD_PROFILE} "${ECG_BUILD_PROFILE}")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(MINIMAL_BUILD ON)
project(ecg_AFib)
